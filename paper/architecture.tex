% Chatbot architecture, simplified. Same drawing as thesis/figures/architecture.tex.
\begin{figure*}[t]
\centering
\begin{tikzpicture}[x=1cm,y=1cm,font=\small,
  box/.style={draw=black!75,rounded corners=1.5pt,fill=white,align=center,text width=3.0cm,minimum height=1.05cm,inner sep=3pt},
  tested/.style={box,line width=1.2pt,fill=black!8},
  flow/.style={-{Stealth[length=1.6mm,width=1.15mm]},line width=.5pt},
  two/.style={{Stealth[length=1.6mm,width=1.15mm]}-{Stealth[length=1.6mm,width=1.15mm]},line width=.5pt},
  opt/.style={two,dashed},
  head/.style={font=\small\bfseries,anchor=west},
  lab/.style={font=\footnotesize,fill=white,inner sep=1.5pt,align=center}
]
  % Preparing knowledge
  \node[head] at (-0.2,1.55) {Preparing knowledge};
  \node[box] (docs) at (1.6,0) {Official documents\\(PDF, DOCX)};
  \node[box] (text) at (5.4,0) {Text versions, checked by us};
  \node[box] (index) at (9.2,0) {Indexing: passages and embeddings};
  \node[box] (store) at (13.0,0) {Search index (database)};
  \node[box,minimum height=.75cm] (faq) at (9.2,1.55) {Our FAQs};
  \draw[flow] (docs) -- (text);
  \draw[flow] (text) -- (index);
  \draw[flow] (index) -- (store);
  \draw[flow] (faq) -- (index);

  % Answering
  \node[head] at (-0.2,-1.45) {Answering};
  \node[box] (client) at (1.6,-2.6) {Web app or Discord bot};
  \node[box] (api) at (5.4,-2.6) {Chat service};
  \node[box] (agent) at (9.2,-2.6) {AI agent (language model)};
  \node[tested] (search) at (13.0,-2.6) {Knowledge-base search and reranking};
  \draw[two] (client) -- (api);
  \draw[two] (api) -- (agent);
  \draw[two] (agent) -- (search);
  \draw[flow] (store) -- (search);

  % Below the answering row
  \node[box] (conv) at (5.4,-5.0) {Saved conversations};
  \node[box] (tools) at (9.2,-5.0) {Faculty data tools: courses, rooms, timetables, announcements};
  \node[tested] (replay) at (13.0,-5.0) {Controlled experiment: one search, then one model call};
  \draw[two] (api) -- (conv);
  \draw[opt] (agent) -- (tools);
  \draw[flow,line width=.9pt] (search) -- (replay);
\end{tikzpicture}
\caption{Chatbot architecture and the parts tested. Official documents become checked text versions and, with our FAQs, are split into passages with embeddings in a search index. The chat service passes each question to the AI agent, which decides when to search the knowledge base and when to call the faculty data tools (dashed). A separate model server computes embeddings and reranking scores. Bold boxes are the parts the controlled experiment tested, where one search runs first and a single model call answers, without the agent or tools.}
\label{fig:architecture}
\end{figure*}
