% Assessment pipeline of the controlled experiment: reference, answer, judge, labels, guard.
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[font=\small,
  st/.style={draw,rounded corners=2pt,align=center,text width=2.45cm,minimum height=3.3cm,inner sep=3pt,fill=white},
  hd/.style={font=\small\bfseries},
  res/.style={draw,rounded corners=2pt,align=center,text width=2.45cm,minimum height=3.3cm,inner sep=3pt,fill=black!20,thick},
  arr/.style={-{Stealth[length=2.2mm]},thick}]
  \node[st] (r) {\textbf{Reference}\\facts a correct answer needs, one per question, quotations checked};
  \node[st,right=0.4cm of r] (a) {\textbf{Answer}\\one per policy, reused when the prompt is identical};
  \node[st,right=0.4cm of a] (j) {\textbf{Judge}\\sees one answer, its question, the reference and the prompt, not the policy};
  \node[st,right=0.4cm of j] (l) {\textbf{Labels}\\one category and three flags per answer};
  \node[res,right=0.4cm of l] (g) {\textbf{Guard}\\a positive label becomes uncertain if the reference failed its check};
  \foreach \p/\q in {r/a,a/j,j/l,l/g} {
    \draw[arr] (\p) -- (\q);
  }
\end{tikzpicture}
\caption[Assessment pipeline of the controlled experiment]{Assessment pipeline of the controlled experiment. All 4 policies are judged against the same reference for each question. The guard is applied after judging and only lowers positive labels.}
\label{fig:assessment-pipeline}
\end{figure}
